\section{Conventions}

\begin{function}[]
  {
    \foo, \foovariant
  }
  \begin{syntax}
    \cs{foo} \oarg{optional arg} \Arg{mandatory arg}
  \end{syntax}
  General form of documentation for commands. Commands that share a
  common behavior/interface may be documented together like this.

  The syntax box gives one example with arguments. Commands that are
  not user-level and take no arguments (variables, constants,
  internal helpers) have no syntax box. A command that is not yet
  documented has no explanation afterwards.

  For internal commands, lists of packages in \TeX\,Live that redefine
  or patch them (\enquote{Modified by}) and that use them
  (\enquote{Used by}) are given after the entry. They come from an
  automated search of \texttt{tex/latex} that excludes the core
  bundles, so they are indicative only.

\end{function}

\begin{function}[TF]
  {
    \foo
  }
  \begin{syntax}
    \cs{fooF} \Arg{text arg} \Arg{false branch}
  \end{syntax}
  Conditionals using the \texttt{TF} convention are documented like
  this.  In \LaTeXe{} commands normally do not indicate the arguments
  they take (\texttt{TF} is one of the exceptions). But any other
  arguments aren't indicated so one has to \enquote{know}.

  In contrast,
  commands from the L3 programming layer always denote their argument
  signature in the name (via the \texttt{:} convention).
  
\end{function}

\begin{function}[int]
  {
    \foo
  }
  Internal commands that should not be relied on are marked in red.
  
\end{function}

\begin{function}[user]
  {
    \foo
  }
  \begin{syntax}
    \cs{foo} 
  \end{syntax}
  %
  Document-level (user) commands are shown in blue. In addition
  commands that are mainly for class and package writer but can also
  be useful in the document preamble or inside the document are also
  marked this way. This means that basically all core commands
  documented in \emph{The LaTeX{} Companion}\cite{tlc3} are shown in
  blue.
  
\end{function}

\begin{function}[EXP]
  {
    \foo
  }
  Fully expandable functions, i.e., those that can be used
  inside \cs{edef} or \cs{expanded} are indicated with a star.

  Most user commands can be safely put into \cs{edef} but then do
  nothing at all (because they are internally protected either
  by \cs{protect} or by being defined with \cs{protected}). Such
  commands are not marked with a star!
  
\end{function}

\subsubsection{Modern commands with extended syntax}

\begin{function}[user]
  {
    \NewDocumentCommand,
    \RenewDocumentCommand,
    \DeclareDocumentCommand,
    \ProvideDocumentCommand
  }
  \begin{syntax}
    \cs{NewDocumentCommand} \meta{cmd} \Arg{arg spec} \Arg{code}
  \end{syntax}
  Defines \meta{cmd} as a robust document command whose arguments are
  described by \meta{arg spec}, e.g., \texttt{m} for a mandatory
  argument, \texttt{o} or \texttt{O\Arg{default}} for an optional
  argument in brackets, \texttt{s} for a star, and so on. Inside
  \meta{code} the arguments are available as \texttt{\#1},
  \texttt{\#2}, etc. The variants differ only in what happens if
  \meta{cmd} is already defined (or not): \cs[no-index]{New\ldots}
  gives an error if it exists, \cs[no-index]{Renew\ldots} gives an
  error if it doesn't, \cs[no-index]{Provide\ldots} silently does
  nothing if it exists, and \cs[no-index]{Declare\ldots} always
  defines it.
  This is the recommended interface for new commands that take
  arguments; for details see \texttt{usrguide}.
\end{function}




\begin{function}[user]
  {
    \NewExpandableDocumentCommand,
    \RenewExpandableDocumentCommand,
    \DeclareExpandableDocumentCommand,
    \ProvideExpandableDocumentCommand
  }
  \begin{syntax}
    \cs{NewExpandableDocumentCommand} \meta{cmd} \Arg{arg spec} \Arg{code}
  \end{syntax}
  Like \cs{NewDocumentCommand} and friends, but \meta{cmd} is made
  expandable, so it can be used inside \cs{edef}, \cs{csname}, or
  similar contexts. In return the \meta{arg spec} is restricted: the
  last argument must be mandatory and some types (such as
  verbatim arguments) are not supported. Only use this if expandability
  is really needed.
\end{function}

\subsubsection{Basic command definitions}

\begin{function}[user]
  {
    \newcommand,
    \renewcommand,
    \providecommand
  }
  \begin{syntax}
    \cs{newcommand} * \meta{cmd} \oarg{num} \oarg{default} \Arg{code}
  \end{syntax}
  Defines \meta{cmd} to take \meta{num} arguments (0--9, default 0),
  referred to as \texttt{\#1}, \ldots, \texttt{\#9} in \meta{code}. If
  \meta{default} is given, the first argument becomes optional (in
  brackets) and \meta{default} is used when it is absent; all further
  arguments are mandatory. \cs{newcommand} gives an error if
  \meta{cmd} is already defined, \cs{renewcommand} if it isn't, and
  \cs{providecommand} makes the definition only if \meta{cmd} is
  undefined.

  Without the star the arguments are \cs{long}, i.e., may contain
  \cs{par}; the starred form produces commands whose arguments can't.
  Commands without arguments are never made \cs{long} (since 2025/11).
  For more complex argument structures use \cs{NewDocumentCommand}.
\end{function}



\begin{function}[]
  {
    \CheckCommand
  }
  \begin{syntax}
    \cs{CheckCommand} * \meta{cmd} \oarg{num} \oarg{default} \Arg{code}
  \end{syntax}
  Takes the same arguments as \cs{newcommand} but defines nothing.
  Instead, it checks that \meta{cmd} currently has exactly that
  definition and issues a warning if not. Default values for optional
  arguments are not compared. Useful in packages before redefining a
  kernel or package command, to detect that some other package has
  already changed it. Only usable in the preamble.
\end{function}



\begin{function}[int]
  {
    \check@command,
    \@check@c,
    \@check@eq
  }
  Internals for \cs{CheckCommand}: \cs{check@command} grabs the
  arguments, \cs{@check@c} defines a scratch command with them, and
  \cs{@check@eq} compares it with the existing definition using
  \cs{ifx} and warns if they differ.
\end{function}



\begin{function}[int]
  {
    \@star@or@long
  }
  \begin{syntax}
    \cs{@star@or@long} \meta{definer}
  \end{syntax}
  Looks ahead for a star, sets \cs{l@ngrel@x} accordingly, resets
  \cs{pr@tectedrel@x} to \cs{relax}, and then calls \meta{definer}
  (e.g., \cs{new@command}). This is how \cs{newcommand} and friends
  implement their star forms.

\end{function}

\subsubsection{Low-level command definitions}

These are \TeX{} primitives; they are listed here because package
code uses them all the time. Unlike \cs{newcommand} they never check
whether the command already exists.

\begin{function}[]
  {
    \def
  }
  \begin{syntax}
    \cs{def} \meta{cmd} \meta{parameter text} \Arg{replacement text}
  \end{syntax}
  Defines \meta{cmd} locally. The \meta{parameter text} consists of
  parameters \texttt{\#1} to \texttt{\#9} (in order), possibly mixed
  with delimiter tokens; e.g., with \texttt{\#1.\#2} as \meta{parameter
  text} the first argument is delimited by a dot. Prefixes \cs{global}, \cs{long}, and \cs{protected} can
  be put in front. The \meta{replacement text} is stored unexpanded.
\end{function}


\begin{function}[]
  {
    \edef
  }
  \begin{syntax}
    \cs{edef} \meta{cmd} \meta{parameter text} \Arg{replacement text}
  \end{syntax}
  Like \cs{def}, but the \meta{replacement text} is fully expanded at
  definition time. Robust \LaTeX{} commands survive this unchanged, but
  fragile ones may break; if the text can contain arbitrary user input
  use \cs{protected@edef} instead.
\end{function}


\begin{function}[]
  {
    \xdef
  }
  \begin{syntax}
    \cs{xdef} \meta{cmd} \meta{parameter text} \Arg{replacement text}
  \end{syntax}
  Same as \cs{global}\cs{edef}. The \LaTeX{} variant that respects
  \cs{protect} is \cs{protected@xdef}. (There is also the primitive
  \cs{gdef}, the same as \cs{global}\cs{def}.)
\end{function}




\begin{function}[]
  {
    \let
  }
  \begin{syntax}
    \cs{let} \meta{cmd} \texttt{=} \meta{token}
  \end{syntax}
  Gives \meta{cmd} the current meaning of \meta{token} (the
  \texttt{=} and one following space are optional). If \meta{token} is
  a macro, \meta{cmd} becomes a copy of it that does not change if
  \meta{token} is later redefined. Note that this doesn't work properly
  for robust commands defined with \cs{DeclareRobustCommand} or with
  an optional argument via \cs{newcommand}; use \cs{NewCommandCopy}
  for them.
\end{function}



\begin{function}[]
  {
    \futurelet
  }
  \begin{syntax}
    \cs{futurelet} \meta{cmd} \meta{token\textsubscript1} \meta{token\textsubscript2}
  \end{syntax}
  Sets \meta{cmd} to the meaning of \meta{token\textsubscript2} (\`a
  la \cs{let}) and then executes \meta{token\textsubscript1} with
  \meta{token\textsubscript2} still in the input. This is the basis
  for look-ahead commands like \cs{@ifnextchar}, which should normally be
  used instead.
\end{function}

\subsubsection{Modern environments with extended syntax}

\begin{function}[user]
  {
    \NewDocumentEnvironment,
    \RenewDocumentEnvironment,
    \DeclareDocumentEnvironment
  }
  \begin{syntax}
    \cs{NewDocumentEnvironment} \Arg{name} \Arg{arg spec} \Arg{begin code} \Arg{end code}
  \end{syntax}
  Defines the environment \meta{name} with arguments described by
  \meta{arg spec}, as for \cs{NewDocumentCommand}. The arguments are
  available in both \meta{begin code} and \meta{end code}. A
  \texttt{b} in the \meta{arg spec} grabs the whole environment body
  as an argument. The variants differ in their handling of existing
  definitions, as for the corresponding command declarations
  (\cs{ProvideDocumentEnvironment} also exists).
\end{function}

\subsubsection{Basic environments}




\begin{function}[user]
  {
    \newenvironment,
    \renewenvironment
  }
  \begin{syntax}
    \cs{newenvironment} * \Arg{name} \oarg{num} \oarg{default} \Arg{begin code} \Arg{end code}
  \end{syntax}
  Defines the environment \meta{name}; arguments are handled as for
  \cs{newcommand} but are only available in \meta{begin code}. To use
  them in \meta{end code} they have to be saved first (or use
  \cs{NewDocumentEnvironment}). The environment is implemented by the
  commands \cs[no-index]{\meta{name}} and \cs[no-index]{end\meta{name}}, so
  \cs{newenvironment} fails if either already exists.
  \cs{renewenvironment} gives an error if the environment is undefined.
\end{function}




\begin{function}[int]
  {
    \renew@environment
  }
  \begin{syntax}
    \cs{renew@environment} \Arg{name} \oarg{num} \oarg{default} \Arg{begin code} \Arg{end code}
  \end{syntax}
  The workhorse behind \cs{renewenvironment} after the star has been
  processed. Use \cs{renewenvironment} instead.
\end{function}

\subsubsection{Support for modern commands and environments with extended syntax}




\begin{function}[TF,user]
  {
    \IfBoolean
  }
  \begin{syntax}
    \cs{IfBooleanTF} \meta{boolean} \Arg{true} \Arg{false}
  \end{syntax}
  Tests if \meta{boolean} is either \cs{BooleanFalse} or
  \cs{BooleanTrue} and selects the corresponding argument for further
  processing.
\end{function}

\begin{function}[]
  {
    \BooleanFalse,
    \BooleanTrue
  }
  The values passed by document command parsers for argument types
  \texttt{s} and \texttt{t} (star or token present or not). They
  should only be tested with \cs{IfBooleanTF}, never by comparing them
  directly.
\end{function}





\begin{function}[TF,user]
  {
    \IfNoValue
  }
  \begin{syntax}
    \cs{IfNoValueTF} \Arg{data} \Arg{true} \Arg{false}
  \end{syntax}
  Tests if \meta{data} is just a \cs{NoValue} (as produced, for
  example, by an optional argument to \cs{NewDocumentCommand} if it is
  not present). Depending on the result the \meta{true} or
  \meta{false} argument is selected for further processing.
\end{function}



\begin{function}[]
  {
    \NoValue
  }
  Special command that indicates that something has no value. Can only
  be used together with tests like \cs{IfNoValueTF}. It is generated
  by the parser for document-level commands and environments when, for
  example, an optional argument is not present.
\end{function}



\begin{function}[TF,user]
  {
    \IfValue
  }
  \begin{syntax}
    \cs{IfValueTF} \Arg{data} \Arg{true} \Arg{false}
  \end{syntax}
  The inverse of \cs{IfNoValueTF}.
\end{function}



\begin{function}[user]
  {
    \ReverseBoolean
  }
  \begin{syntax}
    \cs{NewDocumentCommand}\cs[no-index]{foo}\{>\{\cs{ReverseBoolean}\}s m\}\Arg{code}
  \end{syntax}
  Argument processor for the \meta{arg spec} of
  \cs{NewDocumentCommand} and friends (placed in front of an
  \texttt{s} or \texttt{t} argument after \texttt{>}). It swaps
  \cs{BooleanTrue} and \cs{BooleanFalse}.
\end{function}


\begin{function}[user]
  {
    \SplitArgument
  }
  \begin{syntax}
    >\{\cs{SplitArgument} \Arg{number} \Arg{token}\} m
  \end{syntax}
  Argument processor that splits the argument at each \meta{token}
  into exactly \meta{number}${}+1$ parts, which are passed on as
  separate brace groups (missing parts become \cs{NoValue}). Too many
  occurrences of \meta{token} produce an error.
\end{function}


\begin{function}[user]
  {
    \TrimSpaces
  }
  \begin{syntax}
    >\{\cs{TrimSpaces}\} m
  \end{syntax}
  Argument processor that removes leading and trailing spaces from the
  argument.
\end{function}


\begin{function}[user]
  {
    \ProcessList
  }
  \begin{syntax}
    >\{\cs{SplitList} \Arg{token}\} m
    \cs{ProcessList} \Arg{list} \Arg{function}
  \end{syntax}
  The argument processor \cs{SplitList} turns an argument into a list
  of items separated by \meta{token}. \cs{ProcessList} then applies
  \meta{function} to each item, passing the item as an argument.
\end{function}

\subsection{\LaTeX{} counters and lengths commands}

%-----------------------------------------------
\subsubsection{\LaTeX{} counters}

Each \LaTeX{} counter \meta{name} has a corresponding \TeX{} counter
register (defined internally with \cs{newcount}) with the name
\cs{c@\meta{name}}. All counter assignments are global.

\begin{function}[user]
  {
    \newcounter
  }
  \begin{syntax}
    \cs{newcounter} \Arg{name} \oarg{within}
  \end{syntax}
  Allocates a new \LaTeX{} counter \meta{name}, initialized to zero,
  together with \cs{the\meta{name}} (defined as
  \cs{arabic}\Arg{name}) and an empty \cs{p@\meta{name}} used as a
  prefix by \cs{ref}. Gives an error if the counter already exists. If
  \meta{within} is given, \meta{name} is reset to zero whenever
  \meta{within} is stepped by \cs{stepcounter} or
  \cs{refstepcounter}. Unlike \cs{counterwithin}, this doesn't change
  \cs{the\meta{name}}. Counters are always global.
\end{function}

\begin{function}[user]
  {
    \setcounter,
    \addtocounter,
    \stepcounter
  }
  \begin{syntax}
    \cs{setcounter} \Arg{name} \Arg{integer}
    \cs{addtocounter} \Arg{name} \Arg{integer}
    \cs{stepcounter} \Arg{name}
  \end{syntax}
  Set the counter \meta{name} to \meta{integer}, add \meta{integer}
  to it, or increment it by one. \cs{stepcounter} also resets all
  counters declared to be reset within \meta{name} (via the optional
  argument of \cs{newcounter} or \cs{counterwithin}). \meta{integer}
  can be anything \TeX{} accepts as a number, e.g., \cs{value}\Arg{other}.
  An undefined counter gives an error. The name \texttt{*} denotes the
  current counter, i.e., the one stored in \cs{@currentcounter}.
\end{function}

  \end{syntax}

  tagging) via the socket \texttt{refstepcounter/target}.
\end{function}

  \end{syntax}

\end{function}

\end{function}



\begin{function}[user]
  {
    \counterwithin,
    \counterwithout
  }
  \begin{syntax}
    \cs{counterwithin} * \oarg{format-cmd} \Arg{counter} \Arg{parent-counter}
    \cs{counterwithout} * \oarg{format-cmd} \Arg{counter} \Arg{parent-counter}
  \end{syntax}

  another counter is stepped is also available as the command
  \cs{counterwithin}. In contrast to \cs{newcounter}, it alters
  \cs{the\meta{counter}} to typeset
  \cs{the\meta{parent-counter}}\texttt{.}\meta{format-cmd}\texttt{\{\meta{counter}\}}

  optional argument is not present, it defaults to \cs{arabic}.

  When using the starred form \cs{the\meta{counter}} is not touched,
  which is useful if it needs a more complex definition or is already
  correctly set up. Obviously, in that case the optional argument is
  meaningless and is ignored if given.

  The \cs{counterwithout} command implements the opposite operation;
  without star it sets \cs{the\meta{counter}} to
  \meta{format-cmd}\texttt{\{\meta{counter}\}}.
\end{function}

\subsubsection{\LaTeX{} lengths}

\LaTeX{} doesn't distinguishes between skips (variabl lengths) and
dimens (fixed lengths), the high-level user commands only provide
skips (called \enquote{lengths} in the \LaTeX{} manual.

\begin{function}[user]
  {
    \newlength
  }
  \begin{syntax}
    \cs{newlength} \meta{cmd}
  \end{syntax}
  Allocates a new skip register as \meta{cmd}, initialized to
  \texttt{0pt}. Gives an error if \meta{cmd} is already defined.
\end{function}

\begin{function}[user]
  {
    \setlength,
    \addtolength
  }
  \begin{syntax}
    \cs{setlength} \meta{cmd} \Arg{length}
    \cs{addtolength} \meta{cmd} \Arg{length}
  \end{syntax}
  Sets the length register \meta{cmd} to \meta{length}, or adds
  \meta{length} to it. \meta{cmd} can be any \TeX{} skip or dimen
  register (or parameter such as \cs{parindent}). \meta{length} may
  include stretch and shrink components, e.g., \texttt{6pt plus 2pt},
  but only if \meta{cmd} is a skip; for a dimen the \texttt{plus}
  part would end up being typeset. The assignment is
  local, unlike counter assignments.

  With the \pkg{calc} package, \meta{length} can be an arithmetic expression.
\end{function}


\begin{function}[user]
  {
    \settowidth,
    \settoheight
  }
  \begin{syntax}
    \cs{settowidth} \meta{cmd} \Arg{material}
  \end{syntax}
  Typesets \meta{material} in LR-mode in a temporary box (with
  tagging suspended) and sets the length register \meta{cmd} to the
  natural width (respectively height) of the result. The material is
  not output. \cs{settodepth} is the analogue for the depth.
\end{function}


\begin{function}[]
  {
    \@settodim
  }
  \begin{syntax}
    \cs{@settodim} \meta{box dimension} \meta{register} \Arg{material}
  \end{syntax}
  The common implementation of \cs{settowidth} and friends: typesets
  \meta{material} in a temporary box and assigns its
  \meta{box dimension} (\cs{wd}, \cs{ht}, or \cs{dp}) to
  \meta{register}. Normally the user-level commands are preferable.

\end{function}
\modifiers{chronos}
\publicreaders{}



\begin{function}[]
  {
    \@settopoint
  }
  \begin{syntax}
    \cs{@settopoint} \meta{skip or dimen register}
  \end{syntax}
   This macro takes the contents of the skip or dimen register that is
   supplied as its argument and removes the fractional part to make
   it a whole number of points. This can be used in class files to
   avoid values like |345.4666666pt| when calculating a dimension.
\end{function}

\subsection{Text commands}


\begin{function}[user]
  {
    \ ,
    \space,
  }
  \begin{syntax}
    \cs{\space} \cs{\space} \cs{\space}
  \end{syntax}
  %
  The command \verb*=\ = typesets a space character, same as just
  using a \verb*= =, but it can be used in places where a normal space
  gets ignored, e.g., after a command such as \verb*=\LaTeX\ =. Several
  such commands in a row produce multiple spaces in contrast to using
  several normal spaces in a row, e.g., the example above would produce 3
  spaces.

  The \verb*=\ = command is a \TeX{} primitive for typesetting a space
  and for some operations, such as assigning a \cs{catcode} to the
  space character or displaying a space in \cs{typeout} one needs to
  use \cs{\space} to refer to this character.

  The command \cs{space} is a command that contains a space character
  as its replacement material. For typesetting this is identical to
  using \verb*=\ = but in other cases there are differences:
  \cs{csname}\texttt{ X}\cs{space}\cs{endcsname} produces a macro name with a space
  character in the name. In that position \cs{\space} (as a
  typesetting instruction) would not be allowed. For the same reason
  you need to use \cs{space} and not \cs{\space} inside \cs{typeout}.

\end{function}

\subsubsection{Text emphasis}


\begin{function}[user]
  {
    \emph,
    \strongemph
  }
  \begin{syntax}
    \cs{emph}  \Arg{text}
    \cs{strongemph}  \Arg{text}
  \end{syntax}
  Mark \meta{text} as emphasized text or strongly emphasized
  text. Rendering depends on configuration. \cs{emph} usually
  alternates between italics and normal shape, \cs{strongemph}
  typically uses bold face. Configurable through declarations
  \cs{DeclareEmphSequence} and \cs{DeclareStrongEmphSequence}.
  The corresponding declarations are \cs{em} and \cs{strongem}.
\end{function}


\begin{function}[]
  {
    \DeclareEmphSequence,
    \DeclareStrongEmphSequence
  }
  \begin{syntax}
    \cs{DeclareEmphSequence} \Arg{list of font changes}
  \end{syntax}
  The \meta{list of font changes} is a comma separated list whose
  entries are used when \cs{emph} is called (later entries are used if
  \cs{emph} is nested). An entry is skipped if it doesn't change the
  current font (unless it contains \cs{emforce}). If the nesting is
  deeper than available entries \cs{emreset} is used after the last
  entry and then the sequence is repeated. The declarations are
  additive, e.g.,
  \begin{alltt}
    \cs{DeclareEmphSequence}\{\cs{itshape},\cs{upshape}\cs{scshape},\cs{itshape}\}
  \end{alltt}

  \cs{DeclareStrongEmphSequence} provides the same for \cs{strongemph}.

\end{function}



\begin{variable}[user,deprecated]
  { \eminnershape }
  If \cs{DeclareEmphSequence} hasn't been set up
  then \cs{emph} alternates between italics and \cs{eminnershape} (old
  model).
\end{variable}




\begin{function}[user]
  {
    \emforce,
    \strongemforce
  }
  \begin{syntax}
    \cs{DeclareEmphSequence}\{\cs{itshape},\cs{color}\{red\}\cs{emforce}\}
  \end{syntax}
  Used inside an entry of \cs{DeclareEmphSequence}
  (respectively \cs{DeclareStrongEmphSequence}) to force the use of
  that entry even if it doesn't change the font, e.g., if it only
  changes the color.
\end{function}




\begin{function}[user]
  {
    \emreset,
    \strongemreset
  }
  \begin{syntax}
    \cs{renewcommand}\cs{emreset}\Arg{declarations}
  \end{syntax}
  Declarations used by \cs{emph} (respectively \cs{strongemph}) when
  the nesting is deeper than the entries in the declared sequence;
  afterwards the sequence starts again. By default \cs{emreset}
  selects the upright shape and \cs{strongemreset} bold upright.
\end{function}

\subsubsection{Hooks}

It is a bit of a debate which hook commands are user facing. Most are
really meant for packages or classes.  A few are sometimes useful in
the preamble or even in the document body and those have been marked
\enquote{user} but this is at best a rough indication. For details see
the documentation of \texttt{lthooks}.

\begin{function}[]
  {
    \NewHook,
    \NewReversedHook,
    \NewMirroredHookPair
  }
  \begin{syntax}
    \cs{NewHook} \Arg{hook}
    \cs{NewReversedHook} \Arg{hook}
    \cs{NewMirroredHookPair} \Arg{hook\textsubscript1} \Arg{hook\textsubscript2}
  \end{syntax}
  Declare a new hook, i.e., a named place in the code (executed with
  \cs{UseHook}) to which packages and users can add code with
  \cs{AddToHook}. In a reversed hook the code chunks are executed in
  reverse order, which is useful for \enquote{end} hooks matching a
  \enquote{begin} hook; \cs{NewMirroredHookPair} declares such a pair.
  In package code \meta{hook} can start with \texttt{.}, which is
  replaced by the package name. Hook names must be unique and
  declarations are global.
\end{function}






\begin{function}[]
  {
    \NewHookWithArguments,
    \NewReversedHookWithArguments,

  }

  \end{syntax}

  \cs{UseHookWithArguments} and can be accessed as \texttt{\#1},

\begin{function}[user]
  {
    \AddToHook
  }
  \begin{syntax}
    \cs{AddToHook} \Arg{hook} \oarg{label} \Arg{code}
  \end{syntax}

  (default: the current package name, or \texttt{top-level} in the

  such as \texttt{env/\meta{env}/begin}, \texttt{cmd/\meta{cmd}/before}, or

  in \meta{code} must be doubled if the hook takes arguments; use

\end{function}

  executed after the normal hook code and then discarded. Code added
  this way can't be ordered by rules.
\end{function}

\end{function}

\begin{function}[]

\begin{function}[]

\subsubsection{Sockets}

Sockets are named places in the code whose behavior is determined by
an assigned \enquote{plug}, so that different implementations (e.g.,
with and without tagging) can be switched in and out. For details see
the documentation of \texttt{ltsockets}.

\begin{function}[]
  {
    \NewSocket
  }
  \begin{syntax}
    \cs{NewSocket} \Arg{socket} \Arg{number of inputs}
  \end{syntax}
  Declares a new socket that expects \meta{number of inputs}
  arguments. It automatically gets a plug \texttt{noop}, which
  removes the inputs; sockets with one input also get the plug
  \texttt{identity} (returning the input unchanged), which is then
  the default. Only allowed at top level, not in a group.
\end{function}


\begin{function}[]
  {
    \UseSocket
  }
  \begin{syntax}
    \cs{UseSocket} \Arg{socket} \meta{inputs}
  \end{syntax}
  Executes the plug currently assigned to \meta{socket}, passing it
  the \meta{inputs} (as many brace groups as declared). For
  performance reasons there is no check that the socket exists.
\end{function}




\begin{function}[]
  {
    \NewSocketPlug
  }
  \begin{syntax}
    \cs{NewSocketPlug}  \Arg{socket}  \Arg{plug} \Arg{code}
  \end{syntax}
  Declares \meta{plug} for \meta{socket}: \meta{code} is executed when
  the plug is assigned and the socket is used, with the socket inputs
  available as \texttt{\#1}, \texttt{\#2}, \ldots. Gives an error if
  the plug already exists. The declaration is local.
\end{function}




\begin{function}[user]
  {
    \AssignSocketPlug
  }
  \begin{syntax}
    \cs{AssignSocketPlug} \Arg{socket} \Arg{plug}
  \end{syntax}
  Makes \meta{plug} the active plug for \meta{socket}. Gives an error
  if either doesn't exist. The assignment is local.
\end{function}





\begin{function}[TF,user,EXP]
  {
    \IfSocketExists,
    \IfSocketPlugExists
  }
  \begin{syntax}
    \cs{IfSocketExistsTF} \Arg{socket} \Arg{true} \Arg{false}
    \cs{IfSocketPlugExistsTF} \Arg{socket} \Arg{plug} \Arg{true} \Arg{false}
  \end{syntax}
  Test whether \meta{socket}, respectively \meta{plug} for
  \meta{socket}, has been declared.
\end{function}





\begin{function}[TF,user,EXP]
  {
    \IfSocketPlugAssigned
  }
  \begin{syntax}
    \cs{IfSocketPlugAssignedTF} \Arg{socket} \Arg{plug} \Arg{true} \Arg{false}
  \end{syntax}
  Tests whether \meta{plug} is the plug currently assigned to
  \meta{socket}.
\end{function}



\begin{function}[user]
  {
    \ShowSocket,
    \LogSocket
  }
  \begin{syntax}
    \cs{ShowSocket} \Arg{socket}
  \end{syntax}
  Display information about \meta{socket} (number of inputs, available
  plugs, and the current assignment). \cs{ShowSocket} stops,
  \cs{LogSocket} writes only to the log.
\end{function}


\begin{function}[user]
  {
    \DebugSocketsOn,
    \DebugSocketsOff
  }
  \begin{syntax}
    \cs{DebugSocketsOn} \ldots{} \cs{DebugSocketsOff}
  \end{syntax}
  Turn on (or off) tracing of sockets: declarations, plug assignments,
  and uses of sockets are then reported. The setting is global.
\end{function}

\subsubsection{Loops}

For new code the L3 programming layer offers much better loop
constructs (e.g., \texttt{\textbackslash clist\_map\_inline:nn}); the
loops below are mostly of interest for reading existing code.

\begin{function}[]
  {
    \@whilenum
  }
  \begin{syntax}
    \cs{@whilenum} \meta{\TeX{} boolean expression} \cs{do} \Arg{loop body}
  \end{syntax}
  \meta{\TeX{} boolean expression} is an expression that can follow
  after \cs{ifnum}, e.g., \cs{@tempcnta}\texttt{>5}. While the
  condition is true \meta{loop body} is executed which is supposed to
  alter the condition eventually, e.g., by changing \cs{@tempcnta} in
  this example. The loop is not executed in a group.
\end{function}




\begin{function}[]
  {
    \@whiledim
  }
  \begin{syntax}
    \cs{@whiledim} \meta{\TeX{} boolean expression} \cs{do} \Arg{loop body}
  \end{syntax}
  Same as \cs{@whilenum} but for expressions following \cs{ifdim}.
\end{function}



\begin{function}[]
  {
    \@whilesw
  }
  \begin{syntax}
    \cs{@whilesw} \meta{\TeX{} \enquote{if} conditional} \cs{fi} \Arg{loop body}
  \end{syntax}
  While \meta{\TeX{} \enquote{if} conditional}, e.g.,
  \cs{if@tempswa}, is true execute \meta{loop body}, which needs to
  set it to false at some point to exit the loop.
\end{function}

  {
    \@iwhilenum,
    \@iwhiledim,
    \@iwhilesw
  }
  Internal recursion helpers for \cs{@whilenum}, \cs{@whiledim}, and
  \cs{@whilesw}.
\end{function}




\begin{function}[]
  {
    \@for
  }
  \begin{syntax}
    \cs{@for} \meta{cmd} := \meta{comma list} \cs{do} \Arg{loop body}
  \end{syntax}
  Executes \meta{loop body} for each item of \meta{comma list}, with
  \meta{cmd} defined (locally, via \cs{def}) to the current item. The
  \meta{comma list} is expanded once, so it can be a macro holding the
  list. Spaces around items are not removed and empty items are
  processed. Afterwards the loop variable is \cs{ifx}-equal to \cs{@nnil}.

  Does not work in code where \cs{ExplSyntaxOn} is active because
  there \texttt{:} has a different \cs{catcode}.
\end{function}

\subsubsection{Generating errors, warnings, and info messages}



\begin{function}[]
  {
    \PackageError
  }
  \begin{syntax}
    \cs{PackageError}\Arg{package}\Arg{message}\Arg{help message}
  \end{syntax}
  Generates an error message for package \meta{package} displaying
  \meta{message} on the terminal introduced by \texttt{Package
    \meta{package} Error:}.  Processing is stopped and awaits a
  reaction from the user. If the user replies with pressing \texttt{h}
  the \meta{help message} is shown as well.

  Inside \meta{message} the command \cs{MessageBreak} can be used to
  produce suitable line breaks to structure the message.  Displaying
  and formatting the error message is done using \cs{GenericError}
  internally.
\end{function}


\begin{function}[]
  {
    \ClassError
  }
  \begin{syntax}
    \cs{ClassError}\Arg{class}\Arg{message}\Arg{help message}
  \end{syntax}
  The \cs{ClassError} is like \cs{PackageError} but uses \texttt{Class
    \meta{class} Error:} to introduce the error message.
\end{function}




\begin{function}[]
  {
    \PackageWarning,
    \PackageWarningNoLine,
    \PackageNote,
    \PackageNoteNoLine,
    \PackageInfo,
  }
  \begin{syntax}
    \cs{PackageWarning}\Arg{package}\Arg{message}
  \end{syntax}
  Generates a package warning on the terminal (with or without
  displaying the current line number). The \meta{message} is preceded
  by \texttt{Package \meta{package} Warning:}. The command
  \cs{MessageBreak} can be used to produce suitable line breaks to
  structure the message.  Displaying and formatting the warning
  message is done using \cs{GenericWarning} internally.

  The commands \cs{PackageNote} and \cs{PackageNoteNoLine} only differ
  in that they use \texttt{Package \meta{package} Info:} at the
  beginning, i.e., they show their \meta{message} on the terminal but
  not as a warning; instead they offer \enquote{information} of some
  kind.

  The command \cs{PackageInfo} is like \cs{PackageNote} but writes its
  \meta{message} only to the transcript and not to the terminal.
\end{function}



\begin{function}[]
  {
    \ClassWarning,

    \ClassNote,
    \ClassNoteNoLine,
    \ClassInfo
  }
  \begin{syntax}
    \cs{ClassNoteNoLine}\Arg{class}\Arg{message}
  \end{syntax}
  These commands are the corresponding commands
  for use in document classes instead of packages.
\end{function}

  \end{syntax}

  \cs{PackageWarning}, etc., indented and prefixed so that it lines up
  with the first line (e.g., with \texttt{(\meta{package})}). Outside
  such messages it is \cs{relax}.
\end{function}

\subsection{Font handling in \LaTeX{} (NFSS)}

In NFSS a font is described by five attributes: encoding, family,
series, shape, and size. For details see \texttt{fntguide}.

\subsubsection{Declarations for text and math fonts}



\begin{function}[user]
  {
    \DeclareFontShape
  }
  \begin{syntax}
    \cs{DeclareFontShape} \Arg{encoding} \Arg{family} \Arg{series} \Arg{shape} \Arg{size specs} \Arg{special}
  \end{syntax}
  Declares which external fonts to use for the given combination of
  attributes at which sizes. \meta{size specs} is a list of entries
  such as \texttt{<10> cmr10} or \texttt{<->s*[0.95] fontname},
  possibly using size functions like \texttt{sub} or \texttt{ssub}
  for substitutions. \meta{special} is code executed each time the
  font is loaded (often empty). Usually found in \texttt{.fd} files;
  the \meta{family} must have been declared with \cs{DeclareFontFamily}.
\end{function}




\begin{function}[user]
  {
    \DeclareFixedFont
  }
  \begin{syntax}
    \cs{DeclareFixedFont} \meta{cmd} \Arg{encoding} \Arg{family} \Arg{series} \Arg{shape} \Arg{size}
  \end{syntax}
  Defines \meta{cmd} as a fast switch to one fixed font, bypassing
  the NFSS selection mechanism. Useful for fonts used often in a fixed
  size (e.g., in headers); \meta{cmd} doesn't update
  \cs{baselineskip}, math fonts, or the current font attributes.
\end{function}



\begin{function}[user]
  {
    \DeclareFontFamily
  }
  \begin{syntax}
    \cs{DeclareFontFamily} \Arg{encoding} \Arg{family} \Arg{special}
  \end{syntax}
  Declares \meta{family} in \meta{encoding}. The \meta{special} code
  is executed when any font of the family is loaded, typically to set
  \cs{hyphenchar} or \cs{skewchar}. Usually the first declaration in
  an \texttt{.fd} file.
\end{function}



\begin{function}[user]
  {
    \DeclareFontFamilySubstitution
  }
  \begin{syntax}
    \cs{DeclareFontFamilySubstitution} \Arg{encoding} \Arg{family} \Arg{substitution}
  \end{syntax}
  Attempts to load \texttt{.fd} files for \meta{family} and the
  \meta{substitution} family in \meta{encoding} if not loaded already
  and then substitutes all kind of standard shapes from the
  \meta{substitution} family to \meta{family}. This means that from
  that point onwards a request for a shape in \meta{family} will
  always use the shape from \meta{substitution}.

  The substitution is done with \texttt{ssub}, i.e., silently and only
  recorded in the \texttt{.log} file.
\end{function}



\begin{function}[int]
  {
    \DeclareErrorFont
  }
  \begin{syntax}
    \cs{DeclareErrorFont} \Arg{encoding} \Arg{family} \Arg{series} \Arg{shape} \Arg{size}
  \end{syntax}
  For a fully functional NFSS setup there has to be one
  \cs{DeclareErrorFont} declaration for each \meta{encoding} known to
  NFSS. Its purpose is to select this font shape when a requested font
  shape can't be found and the automatic substitution
  mechanism also doesn't lead to an existing font.

  This declaration should only be used by the kernel or by the few
  packages that provide additional \meta{encoding}s for \LaTeX. It
  \emph{should not} be used by packages that provide additional fonts,
  which is why it is marked as internal in this document!

  Only available in the preamble.
\end{function}



\begin{function}[user]
  {
    \DeclareFontEncoding
  }
  \begin{syntax}
    \cs{DeclareFontEncoding} \Arg{encoding} \Arg{text settings} \Arg{math settings}
  \end{syntax}
  Declares a new font encoding. \meta{text settings} is executed each
  time the encoding is selected in text, \meta{math settings} when it
  is used for a math alphabet. The commands for the encoding-specific
  glyphs are declared separately (e.g., in an \texttt{\meta{enc}enc.def}
  file) and loaded with \pkg{fontenc}. Only available in the preamble.
\end{function}



\begin{function}[int]
  {
    \noaccents@
  }
  Used as \meta{math settings} of encodings (such as \texttt{U})
  without accents. Empty in the kernel; \pkg{amsmath} defines it to
  mark math accents as not available.
\end{function}





\begin{function}[user]

    \LastDeclaredEncoding
  }

  \end{syntax}

\end{function}



\begin{function}[user]
  {
    \DeclareFontSubstitution
  }
  \begin{syntax}
    \cs{DeclareFontSubstitution} \Arg{encoding} \Arg{family} \Arg{series} \Arg{shape}
  \end{syntax}
  Sets the default font shape for \meta{encoding} that is used as the
  last resort when automatic font substitution doesn't find the
  requested shape. These fonts must exist (at least in one size),
  which is checked at \cs{begin}\texttt{\{document\}}. Only available
  in the preamble.
\end{function}




\begin{function}[user]
  {
    \DeclareFontEncodingDefaults
  }
  \begin{syntax}
    \cs{DeclareFontEncodingDefaults} \Arg{text settings} \Arg{math settings}
  \end{syntax}

\end{function}



\begin{function}[user]
  {
    \DeclareEncodingSubset
  }
  \begin{syntax}
    \cs{DeclareEncodingSubset} \Arg{encoding} \Arg{family} \Arg{subset}
  \end{syntax}
  Declares that \meta{family} supports only a subset of
  \meta{encoding} (currently only used for \texttt{TS1}).
  \meta{subset} ranges from \texttt{0} (all glyphs available) to
  \texttt{9} (many missing). A \meta{family} ending in \texttt{-*}

  \texttt{-OsF}. Usually placed in \texttt{.fd} files.
\end{function}



\begin{function}[user]
  {
    \CheckEncodingSubset
  }
  \begin{syntax}
    \cs{CheckEncodingSubset} \meta{use cmd} \Arg{encoding} \Arg{fallback} \Arg{subset} \meta{symbol}
  \end{syntax}

  \meta{symbol} (i.e., its declared subset of \meta{encoding} is

\end{function}

\subsection{Design}

%-----------------------------------------------
\subsubsection{Templates}

Most of the code for templates is written in the L3 programming layer
and is internal, i.e., the commands start with \texttt{\_\_}. Here we only
describe the public functions. For details see the documentation of
\texttt{lttemplates}.

A \emph{template type} describes an abstract design object (e.g.,
\enquote{sectioning}) and the number of arguments it takes; a
\emph{template} implements a type with a set of keys; an
\emph{instance} is a template with concrete key values that can be
used in code.


\subsubsubsection{Declaring templates}

Despite the fact that template declarations have CamelCase names we do
not list them as user facing because they are mainly intended for
programmers defining design solutions. Typically used either in the
\LaTeX{} kernel or in class or package files.

\begin{function}[]
  {
    \NewTemplateType
  }
  \begin{syntax}
    \cs{NewTemplateType} \Arg{type} \Arg{\# of arguments}
  \end{syntax}
  Declares a new template \meta{type}; every template and instance of
  this type takes \meta{\# of arguments} arguments when used.
  A template type can only be declared once!
\end{function}

\begin{function}[]
  {
    \DeclareTemplateInterface
  }
  \begin{syntax}
    \cs{DeclareTemplateInterface} \Arg{type} \Arg{name} \Arg{\# of arguments} \Arg{key spec}
  \end{syntax}
  Declare the interface for a template named \meta{name} of type
  \meta{type}. The \meta{\# of arguments} must match the number
  declared for \meta{type}.
%
  The \meta{key spec} specifies all keys understood by the template,
  their key-type and if applicable, a default value for each, in the
  form \meta{key}\texttt{\,:\,}\meta{key type}\texttt{\,=\,}\meta{default}.

  The declaration overwrites an existing declaration with the same
  \meta{name} without warning.
\end{function}


\begin{function}[user]
  {
    \KeyValue
  }
  \begin{syntax}
    \meta{key2} : \meta{key type} = \cs{KeyValue} \Arg{key1}
  \end{syntax}
  Refers to the value of another key of the same template. Can be
  used as default values for keys, so that one key follows another
  unless set explicitly. Can also be used in \cs{DeclareInstance} or
  \cs{EditInstance} to specify key values there.
\end{function}


\begin{function}[]
  {
    \EditTemplateDefaults
  }
  \begin{syntax}
    \cs{EditTemplateDefaults} \Arg{type} \Arg{name} \Arg{new defaults}
  \end{syntax}
  Replaces the defaults of template \meta{name} by the key--value list
  \meta{new defaults}. This affects instances declared afterwards;
  existing instances are unchanged.
\end{function}



\begin{function}[]
  {
    \DeclareTemplateInterfaceEquivalents
  }
  \begin{syntax}
    \cs{DeclareTemplateInterfaceEquivalents} \Arg{type} \Arg{name} \oarg{date} \Arg{key spec}
  \end{syntax}
  Declares additional keys for template \meta{name} that act as
  equivalents of its base keys, e.g., as an alias or as a key that sets
  several base keys. If \meta{date} is given, the keys are
  deprecated from that date and their use produces a warning. Useful
  for renaming keys without breaking existing instances.
\end{function}


\begin{function}[]
  {
    \DeclareTemplateCode
  }
  \begin{syntax}
    \cs{DeclareTemplateCode} \Arg{type} \Arg{name} \Arg{\# of arguments} \Arg{key bindings} \Arg{code}
  \end{syntax}
  Implements template \meta{name}: \meta{key bindings} maps each key
  of the interface to a variable (e.g.,
  \texttt{indent = \textbackslash l\_mytemplate\_indent\_dim}), and
  \meta{code} is executed when the template is used, with the
  arguments of the type as \texttt{\#1}, \ldots. Key values are
  assigned to the variables before \meta{code} (or at
  \cs{AssignTemplateKeys}).

  The declaration overwrites an existing declaration with the same
  \meta{name} without warning.
\end{function}



\begin{function}[]
  {
    \AssignTemplateKeys
  }
  Used directly (not nested in a macro) inside the \meta{code} of
  \cs{DeclareTemplateCode} to delay the assignment of the key values
  to their variables until this point; otherwise they are assigned
  before \meta{code} starts.
\end{function}



\begin{function}[]
  {
    \SetTemplateKeys,
    \SetKnownTemplateKeys
  }
  \begin{syntax}
    \cs{SetTemplateKeys} \Arg{type} \Arg{name} \Arg{key settings}
  \end{syntax}
  Used inside the \meta{code} of \cs{DeclareTemplateCode} to
  override key values, typically with a key--value list passed by
  the user as an argument. \cs{SetTemplateKeys} gives an error for
  unknown keys, \cs{SetKnownTemplateKeys} stores them in
  \cs{UnusedTemplateKeys}. If used together with
  \cs{AssignTemplateKeys} the latter should come first.
\end{function}



\begin{function}[]
  {
    \SetCurrentKnownTemplateKeys
  }
  \begin{syntax}
    \cs{SetCurrentKnownTemplateKeys} \Arg{key settings}
  \end{syntax}
  Like \cs{SetKnownTemplateKeys} but uses the type and name of the
  template currently being executed. It should be used before any
  other template is called.
\end{function}


\begin{variable}[]
  {
    \UnusedTemplateKeys
  }
  Holds the key--value pairs that the last
  \cs{SetKnownTemplateKeys} or \cs{SetCurrentKnownTemplateKeys} did
  not know, so that they can be passed on to another template.
\end{variable}




\begin{function}[]
  {
    \DeclareTemplateCopy
  }
  \begin{syntax}
    \cs{DeclareTemplateCopy} \Arg{type} \Arg{name} \Arg{from-name}
  \end{syntax}
  Copies the template definition with the name \meta{from-name} to the
  template with name \meta{name}.
%
  Afterwards one can alter defaults or the code implementation. In
  contrast to \cs{DeclareInstanceCopy} only limited usefulness.

  The declaration overwrites an existing declaration with the same
  \meta{name} without warning.
\end{function}

\subsection{Low-level \TeX{} registers and streams}

The allocation commands below allocate registers globally. Use
\cs{newcounter} and \cs{newlength} for document-level counters and
lengths.

%-----------------------------------------------
\subsubsection{\TeX{} registers}

\begin{function}[]
  {
    \newcount,
    \newdimen,
    \newskip,
    \newmuskip,
    \newtoks
  }
  \begin{syntax}
    \cs{newcount} \meta{cmd}
  \end{syntax}
  A new integer (count), dimen, skip, muskip, or toks register is
  allocated and the command \meta{cmd} refers to it. No check is made
  whether \meta{cmd} is already defined (unlike \cs{newlength}). All
  registers available in the engine (normally 32768 or more) can be
  allocated.
\end{function}


\begin{function}[]
  {
    \newbox
  }
  \begin{syntax}
    \cs{newbox} \meta{cmd}
  \end{syntax}
  Allocates a new box register and defines \meta{cmd} (via
  \cs{chardef}) as its number, so it can be used with \cs{setbox},
  \cs{box}, \cs{wd}, etc. For document-level boxes use \cs{newsavebox}.
\end{function}


\begin{function}[]
  {
    \newlanguage
  }
  \begin{syntax}
    \cs{newlanguage} \meta{cmd}
  \end{syntax}
  Allocates a new language number for hyphenation patterns and
  defines \meta{cmd} (via \cs{chardef}) as that number. Normally only
  used by \pkg{babel} and similar packages, or when the format is
  generated.
\end{function}

\begin{function}[]
  {
    \newinsert
  }
  \begin{syntax}
    \cs{newinsert} \meta{cmd}
  \end{syntax}
  Allocates a new insertion class (which uses box, count, dimen, and
  skip registers of the same number) and defines \meta{cmd} as its
  number. Inserts are used, e.g., for footnotes and floats.
\end{function}

\begin{function}[]
  {
    \newmarks
  }
  \begin{syntax}
    \cs{newmarks} \meta{cmd}
  \end{syntax}
  Allocates a new mark class for the e-\TeX{} \cs{marks} primitive.
  In \LaTeX{} documents and packages use \cs{NewMarkClass} instead.
\end{function}



\begin{function}[]
  {
    \newattribute
  }
  \begin{syntax}
    \cs{newattribute} \meta{cmd}
  \end{syntax}
  Allocates a new attribute register (Lua\TeX{} only) and defines
  \meta{cmd} as an \cs{attributedef} referring to it. Attributes
  are attached to nodes and can be inspected in Lua code.
\end{function}

\subsubsection{Handling \TeX{}-level register allocations}

\begin{function}[int]
  {
    \insc@unt,
    \allocationnumber
  }
  Internal counters to manage \TeX{} registers: the last insert
  number allocated (inserts are allocated downwards) and the number of
  the most recently allocated register.
\end{function}

\begin{function}[int]
  {
    \alloc@
  }
  The plain \TeX{} interface for allocating registers, now implemented
  with \cs{e@alloc}. Use the \cs{new...} commands instead.
\end{function}

\begin{function}[int]
  {
    \e@alloc
  }
  \begin{syntax}
    \cs{e@alloc} \meta{type} \meta{definer} \meta{counter} \meta{reserved} \meta{max} \meta{cmd}
  \end{syntax}
  Generic register allocator: increments \meta{counter}, checks it
  against \meta{max} with \cs{e@ch@ck}, and defines \meta{cmd} with

  \meta{type}. Used by all \cs{new...} register commands.
\end{function}

\begin{function}[int]
  {
    \e@alloc@top
  }
  The highest register number available in the current engine
  (e.g., 32767 or 65535).
\end{function}



\begin{function}[int]
  {
    \e@ch@ck
  }
  Helper for \cs{e@alloc} that checks if the register limit is
  reached, skipping register numbers reserved for inserts and
  giving an error if no register is left.
\end{function}



\begin{function}[int]
  {
    \e@insert@top
  }
  The highest register number that can be used for an insertion class
  (255 in engines with a limited number of inserts).
\end{function}



\subsubsection{Manipulating \TeX{}-level registers}


\begin{function}[]
  {
    \advance,
    \divide
  }
  \begin{syntax}
    \cs{advance} \meta{register} \texttt{by} \meta{value}
  \end{syntax}

\subsubsection{\TeX{} input and output streams}

\begin{function}[]
  {
    \newread,
    \newwrite
  }
  \begin{syntax}
    \cs{newwrite} \meta{cmd}
  \end{syntax}

  A new read or write stream is allocated and the command \meta{cmd}
  refers to it. There are only 16 streams of each kind (in some
  engines more read streams), so they are a limited resource.

\end{function}


\begin{function}[]
  {
    \openin,
    \openout
  }
  \begin{syntax}

  \end{syntax}

  \meta{stream}. Without \cs{immediate}, \cs{openout} is delayed until

\end{function}

\begin{function}[]
  {
    \read
  }
  \begin{syntax}
    \cs{read} \meta{stream} \texttt{to} \meta{cmd}
  \end{syntax}
  \TeX{} primitive that reads the next line (or more, if braces are
  unbalanced) from \meta{stream} and defines \meta{cmd} locally with

\end{function}


\begin{function}[]
  {
    \wlog,
    \protected@wlog
  }
  \begin{syntax}
    \cs{wlog} \Arg{message}
  \end{syntax}
  Writes \meta{message} to the transcript file on a line by
  itself. Commands inside \meta{message} are expanded, which may
  result in errors depending on the content. \cs{protected@wlog} is

\end{function}


\begin{function}[]
  {
    \message
  }
  \begin{syntax}
    \cs{message} \Arg{message}
  \end{syntax}
  A \TeX{} primitive that writes \meta{message} to the terminal and
  transcript file. It doesn't produce new lines, so if one is wanted
  one can use \texttt{\textasciicircum\textasciicircum J}. The
  \meta{message} is expanded. In \LaTeX{} code \cs{typeout} is
  usually preferable.
\end{function}

\subsection{Plain \TeX{} leftovers}

\LaTeX\,2.09 started out as a format on top of plain \TeX{} and kept a
number of plain \TeX{} commands. Some of them have been useful in
programming \LaTeXe{} macros, though the L3 programming layer provides
much better alternatives.  A few ended up in regular use in user
documents, but with a few exceptions this is not recommended!


\begin{function}[user]
  {
    \centerline,
    \leftline,
    \rightline
  }
  \begin{syntax}
    \cs{centerline} \Arg{material}
  \end{syntax}
  %
  Produces a \cs{hbox} of size \cs{hsize} and typesets the
  \meta{material} inside, either centered, flush left, or flush
  right. As a simple \cs{hbox} it does not start a paragraph (as
  \cs{mbox} or \cs{makebox} do), and in a list it ignores the
  indentation.

\end{function}

\begin{function}[int, deprecated]
  {
    \@@line
  }
  \begin{syntax}
    \cs{@@line} \Arg{material}
  \end{syntax}
  Command to implement \cs{centerline}, \cs{leftline}, and
  \cs{rightline}.  Deprecated, do not use directly.
\end{function}


\begin{function}[]
  {
    \clap,
    \llap,
    \rlap
  }
  \begin{syntax}
    \cs{rlap} \Arg{material}
  \end{syntax}
  %
  Puts the \meta{material} into an \cs{hbox} of zero width together
  with \cs{hss} on one or both sides. As a result, the material sticks
  out of the box without taking up any size in \LaTeX{}'s eyes. Thus
  one can overprint material or have something sticking out into the
  margin, etc. However, as this is using an \cs{hbox} no paragraph is
  started so if this command is used just in front of a paragraph it
  will appear on a line before the paragraph---not where most people
  would expect it.

  The safer \LaTeX{} equivalent would be
  \cs{makebox}\oarg{position}\texttt{\{0pt\}}\Arg{material} as this would
  always start a paragraph, if necessary.
\end{function}
